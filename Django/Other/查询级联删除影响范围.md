def get_deleted_objects(instance):
    from lib.django_softdelete_mixin.deletion import SoftDeleteCollector as Collector
    from django.db.models.deletion import Collector

    collector = Collector(using="default", origin=instance)
    collector.collect([instance])
    res = dict()
    for qs in collector.fast_deletes:
        if not qs.exists():
            continue
        res.setdefault(qs.model, []).extend([str(_) for _ in qs])
    for model, instances in collector.data.items():
        res.setdefault(model, []).extend([str(_) for _ in instances])
    
    from django.contrib.admin.models import LogEntry
    if res.get(LogEntry, None):
        del res[LogEntry]

    return res

class TestTestSerializer(serializers.Serializer):

        return sorted(
