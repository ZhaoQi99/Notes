### 版本工具函数
```javascript title:utils/version.js
export function geCommitHash() {
  // eslint-disable-next-line no-undef
  return COMMIT_HASH
}

export function geBuildDate() {
  // eslint-disable-next-line no-undef
  return BUILD_DATE
}

export function getVersion() {
  const mainVersion = require('@ROOT/package.json').version
  return `${mainVersion}(${geBuildDate()}.${geCommitHash()})`
}
```

  commitHash =
    require('child_process')
      .execSync('git rev-parse --short=8 HEAD')
      .toString()
      .trim() || ''

  console.log(e)
}

module.exports = {
  ...
  plugins: [
    new webpack.DefinePlugin({

## 后端实现
### Serializer
```python title:serializers/about.py
import re
from pathlib import Path

import yaml
from django.conf import settings
from rest_framework import serializers

from proj import get_version


class AboutSerializer(serializers.Serializer):
    version = serializers.SerializerMethodField()
    change_log = serializers.SerializerMethodField()

    def get_version(self, obj):
        return get_version()

    def get_change_log(self, obj):
        path = Path(settings.PROJECT_DIR) / "CHANGELOG.yml"
        text = path.read_text()
        if settings.LINK_MR_ENABLE == True:
            url = settings.GIT_URL.strip("/ ")
            text = re.sub(
                r"#(\d+)", rf'<a target="view_window" href="{url}/\1">#\1</a>', text
            )
        else:
            text = re.sub(r"\(#[\d# ]+\)", "", text)
        return yaml.load(text, yaml.Loader)
```
